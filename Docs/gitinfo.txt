Команда	Что делает
git status	            Показать, что изменилось
git add имя_файла	    Добавить конкретный файл в "индекс" (подготовить к коммиту)
git add .	            Добавить все изменённые и новые файлы
git add App/	            Добавить всю папку App
git rm --cached имя_файла   Убрать файл из индекса, но оставить на диске
git commit -m "Описание"    Зафиксировать изменения с комментарием
git push origin main	    Отправить коммиты на удалённый репозиторий (GitHub)
git pull origin main	    Загрузить свежие изменения с GitHub
